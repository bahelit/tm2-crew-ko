// -------------------------------------- //
//  KNOCKOUT — Crew rebuild                 //
//  Re-based onto RoundsBase2 (ManiaPlanet 4 / TM2 framework).               //
//  script version: 2.0.0                  //
// -------------------------------------- //
//
// Slowest racer(s) eliminated each round until one winner remains. Emits the
// KO* XmlRpc callback contract consumed by the PyPlanet app (see
// KNOCKOUT_REBUILD_SPEC.md section 3.1 and apps/knockout/callbacks.py).
//
// ============================ RE-BASE NOTES =============================
// Ported from the old domino54 Knockout (which extended the *deprecated* legacy
// RoundsBase / legacy ModeBase). The live chain is RoundsBase2 -> ModeTrackmania
// -> ModeMatchmaking2 -> ModeBase2 (the generic ModeMatchmaking2/ModeBase2), which
// is a DIFFERENT generation than the legacy ModeBase. Concretely that meant:
//   * Round plugs use their Match_* / Rounds_* forms and stack AFTER the base's
//     (ManiaScript stacks ***Label*** defs across #Extends at the +++Label+++ /
//     ---Label--- site, parent first then child). Match_LogVersions is plural
//     (injected by ModeTrackmania); spawning is the ---Rounds_CanSpawn--- override
//     (survivors = IsRacing); "wait for >=2 players" is the Rounds_WaitForPlayers
//     hook; KO_WarmUp(S_WarmUpNb, cap) = S_WarmUpNb warm-up rounds, each ending
//     when the last player finishes (give-up respawns them at the start instead,
//     so a warm-up lap can be redone; cap from S_WarmUpDuration:
//     <0 no cap, >0 seconds, 0 auto); 0 rounds = none;
//     matchmaking off via Match_Settings (MB_Settings_UseDefaultMatchmaking=False).
//   * XmlRpc:: in this chain binds to Libs/Nadeo/XmlRpc2, which RENAMES the legacy
//     senders: CallbackIsAllowed -> CanSendCallback, SendCallbackArray ->
//     SendCallback. Send every KO* callback through those. Never mix in
//     Libs/Nadeo/TrackMania/XmlRpc.Script.txt to recover the old names -- see the
//     note above the includes; it compiles, runs, and drops every callback.
//     The v2 library also boots DISABLED and the v2 chain never enables it (the
//     legacy TM ModeBase used to, from S_UseScriptCallbacks); only the controller
//     can, once, at connect. We call XmlRpc::Enable() in Match_StartServer -- see
//     the long note there. Registering callbacks is not enough on its own.
//   * The legacy ST2:: scores-table column API is gone (that table is a compiled
//     Hud module). The rich live HUD/overlays are rendered by the PyPlanet app;
//     here we only drive Hud.ScoresTable.SetFooterText. Player/slot colors dropped.
//   * Legacy->v2 symbol swaps: MapIsLapRace -> Map.TMObjective_IsLapRace;
//     MB_SectionRoundNb -> MB_GetRoundCount(); MB_StopRound/MB_StopMap are now
//     FUNCTIONS (MB_StopRound()/MB_StopMap()), round-running read is
//     MB_RoundIsRunning(); MB_VictoryMessage is gone (podium is automatic, so the
//     winner is announced via chat); event loops use Events::Valid(Event) (the v2
//     idiom from stock Rounds), not the legacy PassOn; TM:: (TM3) replaces TM2::;
//     domino54 -> our Libs/crew/Notify.
//   * Round-end cutoff uses the base mechanism: RoundsBase2 sets CutOffTimeLimit =
//     GetFinishTimeout() when S_UseAlternateRules. We provide GetFinishTimeout()
//     (the base calls it). "First finisher" is detected with G_FirstFinisherSeen.
//
// ===================== VALIDATE ON A LIVE SERVER (Phase 4) ==============
// Compiles against this server's stock chain. Still smoke-test on a dev server
// (use S_DebugBotsCount and //ko hud):
//   1. Plug stacking fires for Rounds_CanSpawn / Match_StartRound / Match_PlayLoop
//      defined in this grandchild of RoundsBase2 (eliminated bots must not respawn).
//   2. CTmScore.Points stays a writable survival counter here (we += it on each
//      elimination and emit it in KOMatchStandings). If the base overwrites it,
//      move survival tracking to a private per-login variable.
//   3. No double round-stop at the field-of-1 endgame (RoundsBase2's play loop
//      also stops the round; ours calls MB_StopRound() when <=1 remain).
//   4. All 8 KO* callbacks arrive in the PyPlanet log under
//      S_UseLegacyXmlRpcCallbacks = 0 (PyPlanet forces this). THIS IS THE ONE THAT
//      FAILED on 2026-08-02, twice, for two independent reasons: the senders used
//      the legacy lib (fixed), and the v2 lib was never enabled for a script loaded
//      after the controller connected (fixed). Nothing else in a live cup detects a
//      dead callback path -- the mode plays and chats perfectly with PyPlanet deaf.
//      Check //ko hud before trusting any other knockout test.

// Per-round state shared across the Match_* plug functions. In the v2 plug
// system each ***Match_*** block compiles to a SEPARATE function, so a plain

// They are declared `for This` (mode-persistent) at each use site instead:
//   Integer[] RoundFinishTimes     — players' finish times, slowest last

// ---------------------------------- //
// XmlRpc — turn the library on, THEN register the KO* callback contract.
//
// Libs/Nadeo/XmlRpc2 boots DISABLED (LibXmlRpc2Common_IsEnabled defaults False) and
// CanSendCallback() gates every send on it. Nothing in the ModeBase2 chain enables
// it: the only switch is the controller calling the XmlRpc.EnableCallbacks script
// method. PyPlanet sends that exactly once, inside GbxRemote.connect(), against
// whichever script happened to be running then -- and never again. So every mode
// script loaded LATER (which is what //cup on and //botn do) starts up with the
// whole library off, and both our KO* callbacks and Nadeo's stock ones are dropped
// before they leave the server. The mode plays and chats a flawless knockout with
// the controller completely deaf. Enable it ourselves so the mode does not depend
// on when the controller connected. (The legacy TM ModeBase did exactly this via
// S_UseScriptCallbacks; the v2 chain simply dropped the call and never replaced it.)
// The PyPlanet app re-arms it after each script load too -- belt and braces, since
// that side also covers the stock modes we do not own.
if (!XmlRpc::IsEnabled()) XmlRpc::Enable();

XmlRpc::RegisterCallback("KOPlayerAdded", """
* Data : An array with the login of the player added to the match
* Example : ["Login"]
* Note : This callback is sent when Knockout begins.
""");

XmlRpc::RegisterCallback("KOPlayerRemoved", """
* Data : An array with the login of the player removed from the match
* Example : ["Login"]
* Note : This callback is sent when a player is knocked out.
""");

XmlRpc::RegisterCallback("KOSendWinner", """
* Data : An array with the login of the match winner
* Example : ["Login"]
* Note : This callback is sent when the match is over.
""");

XmlRpc::RegisterCallback("KOMatchStandings", """
* Data : An array of "Login:Points" entries, one per player, for the finished map.
* Example : ["winner_login:3", "second_login:2", "third_login:1", "last_login:0"]
* Note : Sent at map end. Points are the Knockout survival score (highest = winner).
""");

XmlRpc::RegisterCallback("KORoundOrder", """
* Data : An ordered array of "Login:Rank:Cps:Time:Finished" entries for the
*        players still racing this round (Rank 1 = leader, worst is last).
* Example : ["alice:1:3:42000:1", "bob:2:2:-1:0", "cara:3:1:-1:0"]
* Note : Sent live during a round when the order changes. Finished is 1/0.
""");

XmlRpc::RegisterCallback("KORoundStart", """
* Data : ["CurrentRound:TotalRounds"] for the round that just began.
* Example : ["3:5"]  (round 3 of 5; TotalRounds is 0 when unbounded)
""");

XmlRpc::RegisterCallback("KOShieldAwarded", """
* Data : An array with the login of the player who earned a shield.
* Example : ["Login"]
""");

XmlRpc::RegisterCallback("KOShieldUsed", """
* Data : An array with the login of the player whose shield was spent.
* Example : ["Login"]
""");

// Warm-up before the knockout begins.
// S_WarmUpNb = number of warm-up rounds (0 skips; e.g. BOTN TA practice already served).
// Every warm-up round ends as soon as the last player has finished. Give-up is a
// restart here, not a withdrawal: the player respawns at the start and the round
// waits for them (the lib's finish timeout after the first finisher, and the cap
// below, both still apply).
// S_WarmUpDuration (seconds) only caps how long a round may drag on:
//   < 0  → no cap: the round runs until everyone is done (BOTN default)
//   == 0 → cap derived from the author time
//   > 0  → cap of that many seconds (cup Friday uses 120 = 2 minutes per round)
// The PyPlanet app sets S_WarmUpNb / S_WarmUpDuration per event (BOTN / cup presets).
//
// Use KO_WarmUp (not stock MB_WarmUp): the base warm-up loop only Valid()s events and
// never copies finish times into Score.BestRace, so AwardWarmUpShield found nobody.
// KO_WarmUp tracks G_WarmUpBestTimes for the shield earn path while keeping WarmUp::
// start/end so clients still get the stock warm-up UI.
//
// Read AFTER the "New match" settle above (not at the top of StartMap): on the BOTN
// handoff the controller switches into this script and then pushes S_WarmUpNb onto the
// running mode. That brief pause guarantees the freshly-pushed count is in place before
// warm-up reads it -- reading it at the very top raced the push and could give zero
// warm-up laps. See apps/knockout/botn.py (_switch_to_knockout).
declare Integer WarmUpCap = -1;
if (S_WarmUpDuration > 0) {
	WarmUpCap = S_WarmUpDuration * 1000;
} else if (S_WarmUpDuration == 0) {
	WarmUpCap = 0;
}
KO_WarmUp(S_WarmUpNb, WarmUpCap);
UpdatePlayerColors();

// ---------------------------------- //
// Wait for enough players (base injects this in Match_StartRound)
// ---------------------------------- //
***Rounds_WaitForPlayers***
***
WaitForPlayers(C_RequiredPlayersNb);

// Seed the knockout roster once real players + bots have actually joined
// (Match_StartMatch/StartMap are too early — fake users are requested later).
// Reseed when: a new match started (G_NewMatch set in Match_StartMatch), OR none
// of the roster's logins are still present (empty/stale roster — e.g. after a
// //skip that swapped in new players/bots). If any roster survivor is still here
// we do NOT reseed, so survivors carry across maps within a multi-map match.
declare Integer RosterPresent;
RosterPresent = 0;
foreach (Player in Players) if (G_PlayersRacing.exists(Player.User.Login)) RosterPresent += 1;
if (G_NewMatch || RosterPresent == 0) {
	StartKnockout();
	G_NewMatch = False;
}

***

}

***Match_PlayLoop***

	}
	// ---------------------------------- //
	// GiveUp
	else if (Event.Type == CTmModeEvent::EType::GiveUp) {

		Notify::SendChat(["Practice round %1: no eliminations.", TL::ToText(G_RoundsPlayed + 1)]);
	}
	else {
		declare Integer KnockedCount;
		KnockedCount = 0;

		// Knock all DNF (shields never save a DNF)
		foreach (Player in Players) {
			declare Boolean HasFinishedTrack for Player;
			if (GetRacingPlayersAmount() > 1 && !HasFinishedTrack && IsRacing(Player.User)) {
				KnockedCount += 1;
				KnockoutPlayer(Player, True);
			}
		}

		if (KnockedCount == 0) {
			// Everyone finished, but someone gave up
			if (RoundGiveUpsAmount > 0) {
				Notify::SendChat(["Noone is knocked out: %1 player(s) gave up!", TL::ToText(RoundGiveUpsAmount)]);
			}
			// Knock slowest player(s) if all finished and noone gave up
			else if (GetRacingPlayersAmount() > 1 && RoundFinishTimes.count > 0) {
				declare Integer KnocksRemaining;
				KnocksRemaining = 1;
				if (S_DoubleKnockUntil > 0 && GetRacingPlayersAmount() > S_DoubleKnockUntil && GetRacingPlayersAmount() >= 3) KnocksRemaining = 2;

				while (KnocksRemaining > 0 && RoundFinishTimes.count > 0 && GetRacingPlayersAmount() > 1) {
					declare Integer SlowestTime = RoundFinishTimes[RoundFinishTimes.count - 1];

					declare CTmPlayer SlowestPlayer;
					SlowestPlayer = Null;
					foreach (Player in Players) {
						// Match the finisher by their snapshotted round race (Score.BestRace),
						// NOT Player.CurRace: Match_EndRound runs after TM::WaitRaceAll() (above),
						// which ends every race and clears CurRace, so CurRace.Time no longer holds
						// the finish time -- nothing would match, nobody is knocked out, and the
						// knockout loops rounds forever without ever reaching one winner (so
						// Match_EndMap / KOMatchStandings never fire). BestRace is the copy taken at
						// finish in Match_PlayLoop, from the same CurRace.Time that filled
						// RoundFinishTimes, so it matches exactly.
						if (IsRacing(Player.User) && Player.Score != Null
								&& Player.Score.BestRace != Null
								&& Player.Score.BestRace.Time == SlowestTime) {
							SlowestPlayer = Player;
							break;
						}
					}

					declare Removed = RoundFinishTimes.remove(SlowestTime);
					if (SlowestPlayer != Null) {
						// A shield saves only this player and absorbs one knock slot
						// (no cascade to the next-slowest). DNF/give-up never spend shields.
						if (TryConsumeShield(SlowestPlayer)) {
							KnocksRemaining -= 1;
						}
						else {
							KnockoutPlayer(SlowestPlayer, True);
							KnocksRemaining -= 1;
						}
					}
				}
			}
		}
	}
}

/** Live running order, throttled to fire only when it actually changes. */
Void XmlRpc_KORoundOrder() {
	if (!XmlRpc::CanSendCallback("KORoundOrder")) return;

	declare Text[] Data;
	declare Text Sig = "";
	declare Integer Rank = 0;
	foreach (Score in Scores) {
		if (Score.User != Null && G_PlayersRacing.exists(Score.User.Login)) {
			Rank += 1;
			declare Integer Cps = 0;
			declare Integer RaceTime = -1;
			if (Score.BestRace != Null) {
				Cps = Score.BestRace.Checkpoints.count;
				RaceTime = Score.BestRace.Time;
			}
			declare Integer Finished = 0;
			declare CTmPlayer KOPlayer = TM::GetPlayer(Score.User.Login);
			if (KOPlayer != Null) {
				declare Boolean HasFinishedTrack for KOPlayer;
				if (HasFinishedTrack) Finished = 1;
			}
			Data.add(Score.User.Login ^ ":" ^ Rank ^ ":" ^ Cps ^ ":" ^ RaceTime ^ ":" ^ Finished);
			Sig ^= Score.User.Login ^ Finished ^ ";";
		}
	}

	// Skip the send when nothing relevant changed since last time.
	if (Sig == G_PrevRoundOrderSig) return;
	G_PrevRoundOrderSig = Sig;
	XmlRpc::SendCallback("KORoundOrder", Data);
}

/** Warm-up that records best finish times for shield awards.
    Stock MB_WarmUp only Valid()s events and never stores BestRace, so the
    warm-up-fastest shield path always saw an empty field. This mirrors the base
    warm-up UI (WarmUp::Start/StartRound/End) while filling G_WarmUpBestTimes.

    _CapMillisec is a BACKSTOP, not the round length: a warm-up round is over as
    soon as every player has finished or given up. WarmUp::StartRound is always
    called in round mode (-1) because that is the only mode where the lib parks a
    player after their run (finish -> TM::EndRace, give-up -> TM::WaitRace, neither
    respawned) and ends the round on `PlayersRacing.count <= 0`. Passing a real
    time limit flips the lib into time-limited mode, which respawns everyone
    forever and disables that check -- that is why the Friday cup warm-up only
    ended when the 2-minute clock expired. The cap is enforced here instead.

    Give-up is the one part of that parking we undo: a warm-up lap is practice, so
    pressing give-up must put the player back on the start line, not sit them out
    for the rest of the round. See the LibWU3_CanSpawn re-arm in the event loop. */
Void KO_WarmUp(Integer _NbOfWarmUp, Integer _CapMillisec) {
	if (_NbOfWarmUp <= 0) return;

	declare Cap = _CapMillisec;
	if (Cap == 0) {
		declare ObjectiveNbLaps = Map.TMObjective_NbLaps;
		if (ObjectiveNbLaps <= 0 || !Map.TMObjective_IsLapRace) ObjectiveNbLaps = 1;
		declare MaxTime = Map.TMObjective_AuthorTime / ObjectiveNbLaps;
		Cap = 5000 + MaxTime + (MaxTime / 6);
	}

	declare PrevUISequence = UIManager.UIAll.UISequence;
	G_WarmUpBestTimes.clear();

	WarmUp::Start();
	declare WarmUpPlayedNb = 0;
	while (!WarmUp::Finished() && WarmUpPlayedNb < _NbOfWarmUp && !ServerShutdownRequested && !MatchEndRequested) {
		WarmUp::StartRound(WarmUpPlayedNb + 1, _NbOfWarmUp, -1);
		// StartRound spends 3s on the "Warmup: x/y" message then 3s on the spawn
		// countdown before anyone can drive, so the cap starts from that point.
		// CutOffTimeLimit is left alone (-1 in round mode) so the lib can still arm
		// its own straggler timeout when the first player finishes.
		declare RoundDeadline = -1;
		if (Cap > 0) RoundDeadline = Now + 6000 + Cap;
		while (!WarmUp::Finished() && !WarmUp::RoundFinished() && !ServerShutdownRequested && !MatchEndRequested) {
			if (RoundDeadline > 0 && Now >= RoundDeadline) break;
			MB_Yield();

			foreach (Event in PendingEvents) {
				declare Processed = Events::Valid(Event);
				// Give-up during a warm-up lap = restart, not "I am done". The lib
				// spawns each player once per round (StartRound sets LibWU3_CanSpawn
				// False after spawning) and its own Yield() below answers GiveUp with
				// TM::WaitRace, so without this the player is parked for the rest of
				// the round -- the practice lap they wanted to redo is over for them.
				// Re-arming the flag makes the lib's "spawn players joining during the
				// warm up" block put them back on the start line on the next Yield.
				// LibWU3_CanSpawn is `for Score` and already exists, so ASSIGN to it
				// (the declare initializer is ignored) -- same idiom as Rounds_CanSpawn.
				// The round can still not hang: the lib arms its finish timeout once
				// anyone finishes, and RoundDeadline caps the round regardless.
				if (Event.Type == CTmModeEvent::EType::GiveUp) {
					if (Event.Player != Null && Event.Player.Score != Null) {
						declare LibWU3_CanSpawn for Event.Player.Score = True;
						LibWU3_CanSpawn = True;
					}
				}
				// Record finish times only — WarmUp owns race lifecycle.
				if (Event.Type == CTmModeEvent::EType::WayPoint && Event.IsEndRace) {
					if (Event.Player != Null && Event.Player.User != Null && Event.Player.CurRace != Null) {
						declare Integer T = Event.Player.CurRace.Time;
						if (T > 0) {
							declare Text Login = Event.Player.User.Login;
							if (!G_WarmUpBestTimes.existskey(Login) || T < G_WarmUpBestTimes[Login]) {
								G_WarmUpBestTimes[Login] = T;
							}
							// Also keep BestRace when possible (harmless for later fallback).
							if (Event.Player.Score != Null) {
								if (Event.Player.Score.BestRace == Null
										|| Event.Player.Score.BestRace.Compare(Event.Player.CurRace, CTmResult::ETmRaceResultCriteria::Time) <= 0) {
									Event.Player.Score.BestRace = Event.Player.CurRace;
								}
							}
						}
					}
				}
			}

			WarmUp::Yield();
		}
		WarmUp::EndRound();
		WarmUpPlayedNb += 1;
		MB_Yield();
	}
	WarmUp::End();

	UIManager.UIAll.UISequence = CUIConfig::EUISequence::EndRound;
	UIManager.UIAll.BigMessage = _("End of warmup, match starting...");
	CutOffTimeLimit = Now + 4000;
	while (Now < CutOffTimeLimit && !ServerShutdownRequested && !MatchEndRequested) {
		MB_Yield();
	}
	CutOffTimeLimit = -1;
	UIManager.UIAll.BigMessage = "";
	UIManager.UIAll.UISequence = PrevUISequence;
}

/** Give the player with the fastest warm-up race a one-time shield (if enabled).
    Must be defined before StartKnockout (ManiaScript resolves callees top-down).
    Prefers G_WarmUpBestTimes filled by KO_WarmUp; falls back to Score.BestRace. */
Void AwardWarmUpShield() {
	if (!S_EnableShields) return;
	declare CTmPlayer BestPlayer;
	BestPlayer = Null;
	declare Integer BestTime;
	BestTime = -1;

	// Primary: times tracked during KO_WarmUp (look up by login on the player list).
	foreach (Player in Players) {
		if (Player.User == Null) continue;
		declare Text Login = Player.User.Login;
		if (!G_WarmUpBestTimes.existskey(Login)) continue;
		declare Integer T = G_WarmUpBestTimes[Login];
		if (T <= 0) continue;
		if (BestTime < 0 || T < BestTime) {
			BestTime = T;
			BestPlayer = Player;
		}
	}

	// Fallback: Score.BestRace if something else populated it.
	if (BestPlayer == Null) {
		foreach (Player in Players) {
			if (Player.Score == Null || Player.Score.BestRace == Null) continue;
			declare Integer T = Player.Score.BestRace.Time;
			if (T <= 0) continue;
			if (BestTime < 0 || T < BestTime) {
				BestTime = T;
				BestPlayer = Player;
			}
		}
	}

	if (BestPlayer != Null) {
		declare Integer ShieldTokens for BestPlayer;
		if (ShieldTokens < 1) {
			ShieldTokens = 1;
			XmlRpc_KOShieldAwarded(BestPlayer);
			Notify::SendChat(["$<%1$> earns a shield for the fastest warm-up!", BestPlayer.User.Name]);
		}
	}
}

/// Start a new Knockout (seed the racing set + shields).
Void StartKnockout() {
	G_PlayersRacing.clear();
	G_RoundsPlayed = 0;
	// Logins pre-granted a shield this match (e.g. the BOTD fastest-practice time).
	declare Text[] PreShieldLogins = TL::Split(", ", S_PreShieldLogins);
	foreach (Player in Players) {
		G_PlayersRacing.add(Player.User.Login);
		// Fresh knockout: zero the survival score (match-points bucket drives the
		// scoreboard Points column — see Match_StartServer SaveInScore).
		if (Player.Score != Null) Scores::SetPlayerMatchPoints(Player.Score, 0);
		declare Integer ShieldTokens for Player;
		ShieldTokens = 0;
		if (S_EnableShields && PreShieldLogins.exists(Player.User.Login)) {
			ShieldTokens = 1;
			XmlRpc_KOShieldAwarded(Player);
		}
		XmlRpc_KOPlayerAdded(Player);
	}
	// After warm-up (KO_WarmUp runs before WaitForPlayers → StartKnockout), the
	// fastest warm-up finish in G_WarmUpBestTimes earns a one-time save. Tokens
	// were zeroed above so this is the primary earn path for Friday cups.
	AwardWarmUpShield();
}

/** Give a player a one-time shield (capped at one), if shields are enabled. */
Void AwardShield(CTmPlayer _Player) {
	if (!S_EnableShields || _Player == Null) return;
	declare Integer ShieldTokens for _Player;
	if (ShieldTokens < 1) {
		ShieldTokens = 1;
		XmlRpc_KOShieldAwarded(_Player);
		Notify::SendChat(["$<%1$> earns a shield!", _Player.User.Name]);
	}
}

/** Spend a player's shield to save them from a knockout, if they have one. */
Boolean TryConsumeShield(CTmPlayer _Player) {
	if (!S_EnableShields || _Player == Null) return False;
	declare Integer ShieldTokens for _Player;
	if (ShieldTokens > 0) {
		ShieldTokens -= 1;
		XmlRpc_KOShieldUsed(_Player);
		Notify::SendChat(["$<%1$> uses a shield and survives!", _Player.User.Name]);
		return True;
	}
	return False;
}

/// Whether the player is still racing.
Boolean IsRacing(CUser _User) { return (_User != Null && G_PlayersRacing.exists(_User.Login)); }

/// Amount of racing players.
Integer GetRacingPlayersAmount() { return G_PlayersRacing.count; }

/// Update the scores table footer text.
Void UpdateScoresTableFooter() {
	declare TotalRoundsNb = S_RoundsPerMap;
	if (Map.TMObjective_IsLapRace || S_RoundsPerMap <= 0) TotalRoundsNb = 1;

	declare RoundsMessage = MB_GetRoundCount()^" / "^TotalRoundsNb;
	if (TotalRoundsNb <= 1) RoundsMessage = TL::ToText(MB_GetRoundCount());

	declare DoubleKOMessage = "";
	if (S_DoubleKnockUntil >= 3 && GetRacingPlayersAmount() > S_DoubleKnockUntil) DoubleKOMessage ^= TL::ToText(S_DoubleKnockUntil);
	else DoubleKOMessage = _("|FxMotionBlur|Off");

	if (Hud != Null && Hud.ScoresTable != Null) Hud.ScoresTable.SetFooterText(TL::Compose("%1 %2	%3"^GetRacingPlayersAmount()^"	DoubleKO: %4", _("Round"), RoundsMessage, _("Remaining players: "), DoubleKOMessage));

	declare SpectatorsAmount = Spectators.count + Players.count - GetRacingPlayersAmount();
	ModeStatusMessage = TL::Compose(C_ModeStatusMessage, TL::ToText(GetRacingPlayersAmount()), TL::ToText(SpectatorsAmount));
}

/** Knock a player out of the game (survival points to the others). */
Void KnockoutPlayer(CTmPlayer _Player, Boolean _UseMessage) {
	if (_Player != Null && IsRacing(_Player.User)) {
		declare Removed = G_PlayersRacing.remove(_Player.User.Login);

		MB_SortScores(CTmMode::ETmScoreSortOrder::BestRace_CheckpointsProgress);
		UpdateScoresTableFooter();

		XmlRpc_KOPlayerRemoved(_Player);
		if (_UseMessage) Notify::SendChat(["$<%1$> knocked out!", _Player.User.Name]);

		// Survival point for every remaining racer. Go through the Scores library
		// (not a raw Score.Points write) so its internal match-points counter tracks
		// the value — otherwise the library's lifecycle Clear/AffectRoundToMap resets
		// Score.Points back to its own (zero) counter around the podium/next map.
		foreach (Score in Scores) if (G_PlayersRacing.exists(Score.User.Login)) Scores::AddPlayerMatchPoints(Score, 1);
	}
	else return;
}

/// Remove all disconnected players from the racing array.
Void RemoveDisconnected() {
	declare Text[] ConnectedLogins;
	ConnectedLogins.clear();
	foreach (Player in Players) ConnectedLogins.add(Player.User.Login);

	declare Text[] AliveLogins;
	AliveLogins.clear();
	foreach (Login in G_PlayersRacing) AliveLogins.add(Login);

	foreach (Login in AliveLogins)
		if (!ConnectedLogins.exists(Login)) {
			declare Removed = G_PlayersRacing.remove(Login);
		}
}

/** Wait for enough players to launch a round (runs the TimeAttack minigame). */
Void WaitForPlayers(Integer _MinimumNbPlayers) {
	declare HasWaited = False;
	declare WasHold = False;

	if (S_AdminHoldStart) {
		UIManager.UIAll.BigMessage = "";
		UIManager.UIAll.StatusMessage = _("Waiting for start confirmation by admin");
	}

	while (!ServerShutdownRequested && !MatchEndRequested && (Players.count < _MinimumNbPlayers || S_AdminHoldStart)) {
		UIManager.UIAll.UISequence = CUIConfig::EUISequence::None;
		UIManager.UIAll.BigMessage = _("Waiting for players...");
		MB_Yield();
		HasWaited = True;
		WasHold = S_AdminHoldStart;
		---TimeAttack---
	}

	if (HasWaited) {
		TM::WaitRaceAll();
		UIManager.UIAll.BigMessage = _("A new player joins the game.");
		if (!ServerShutdownRequested && !MatchEndRequested) MB_Sleep(5000);
	}

	if (WasHold) UIManager.UIAll.BigMessage = _("Competition is starting!");

	UIManager.UIAll.StatusMessage = "";
	UIManager.UIAll.BigMessage = "";
}

/// Set mode pause (FFA Time Attack minigame).
Void DoPause() {
	UIManager.UIAll.UISequence = CUIConfig::EUISequence::None;
	UIManager.UIAll.BigMessage = TL::Compose("$f90%1", _("Pause"));
	UIManager.UIAll.StatusMessage = _("Competition has been paused for a while");
	UpdatePlayerColors();

	declare HasWaited = False;

	while (!ServerShutdownRequested && !MatchEndRequested && S_AdminSetPause) {
		MB_Yield();
		HasWaited = True;
		---TimeAttack---
	}

	if (HasWaited) {
		TM::WaitRaceAll();
		UIManager.UIAll.StatusMessage = "";
		UIManager.UIAll.BigMessage = _("Competition resumed!");
		if (!ServerShutdownRequested && !MatchEndRequested) MB_Sleep(5000);
	}

	UpdatePlayerColors();
	MB_SortScores(CTmMode::ETmScoreSortOrder::TotalPoints);
	UIManager.UIAll.BigMessage = "";
	UIManager.UIAll.StatusMessage = "";
}

/** Time left to finish the round after the first player. Called by RoundsBase2's
 *	Match_StartRound when S_UseAlternateRules is on. */
Integer GetFinishTimeout() {
	declare FinishTimeout = 0;

	if (S_FinishTimeout >= 0) {
		FinishTimeout = S_FinishTimeout * 1000;
	} else {
		FinishTimeout = 5000;
		if (Map.TMObjective_IsLapRace && NbLaps > 0 && Map.TMObjective_NbLaps > 0) {
			FinishTimeout += ((Map.TMObjective_AuthorTime / Map.TMObjective_NbLaps) * NbLaps) / 6;
		} else {
			FinishTimeout += Map.TMObjective_AuthorTime / 6;
		}
	}

	if (S_UseAlternateRules) {
		if (Map.TMObjective_IsLapRace && NbLaps > 0 && Map.TMObjective_NbLaps > 0) {
			return G_RoundStartTime + ((Map.TMObjective_AuthorTime / Map.TMObjective_NbLaps) * NbLaps) + FinishTimeout;
		} else {
			return G_RoundStartTime + Map.TMObjective_AuthorTime + FinishTimeout;
		}
	} else {
		return Now + FinishTimeout;
	}

	return Now + 15000;
}

Boolean RequestLayerFromXml(Text _XmlPath) {
	if (G_UI_RequestId != NullId && Http.Requests.existskey(G_UI_RequestId)) Http.Destroy(Http.Requests[G_UI_RequestId]);
	G_UI_RequestId = NullId;
	G_UI_RequestEndTime = -1;
	if (!Http.IsValidUrl(_XmlPath)) return False;

	declare Req <=> Http.CreateGet(_XmlPath);
	if (Req == Null) return False;
	G_UI_RequestId = Req.Id;
	G_UI_RequestEndTime = Now + C_UI_RequestTimeout;
	return True;
}

Boolean WaitLayerFromXml() {
	if (Now >= G_UI_RequestEndTime) return False;
	if (G_UI_RequestId == NullId) return False;
	if (!Http.Requests.existskey(G_UI_RequestId)) return False;
	if (Http.Requests[G_UI_RequestId].IsCompleted) return False;
	return True;
}

Boolean SetLayerFromXml() {
	if (!Http.Requests.existskey(G_UI_RequestId)) return False;

	declare Req <=> Http.Requests[G_UI_RequestId];
	if (!Req.IsCompleted || Req.StatusCode != 200) return False;

	Layers::Update("Custom", Req.Result);

	Http.Destroy(Req);
	G_UI_RequestId = NullId;
	G_UI_RequestEndTime = -1;
	return True;
}

Void SetCustomLayerFromXml(Text _Path) {
	if (_Path == "") return;
	declare Loaded = RequestLayerFromXml(_Path);
	if (Loaded) {
		while (WaitLayerFromXml()) yield;
		Loaded = SetLayerFromXml();
	}
}
